from typing import List, Tuple, Optional, Dict, Any

Time = int  # minutes since midnight
Interval = Tuple[Time, Time]


def _merge_intervals(intervals: List[Interval]) -> List[Interval]:
    if not intervals:
        return []

    intervals = sorted(intervals)
    merged = [intervals[0]]

    for start, end in intervals[1:]:
        last_start, last_end = merged[-1]

        if start <= last_end:  # overlap or adjacent
            merged[-1] = (last_start, max(last_end, end))
        else:
            merged.append((start, end))

    return merged


def _apply_buffer(intervals: List[Interval], buffer: int) -> List[Interval]:
    return [(start - buffer, end + buffer) for start, end in intervals]


def _clamp(interval: Interval, bounds: Interval) -> Optional[Interval]:
    start = max(interval[0], bounds[0])
    end = min(interval[1], bounds[1])
    return (start, end) if start < end else None


def recommend_slots(
    working_hours: Interval,
    busy_intervals: List[Interval],
    duration: int,
    buffer: int = 0,
    candidate_window: Optional[Interval] = None,
    max_results: int = 5,
    persona: str = "jane",  # NEW
) -> Any:
    """
    Persona-aware slot recommendation.

    Jane → returns List[Interval]
    Mo   → returns Dict with slots + explanations
    """

    explain = persona == "mo"
    explanations = []

    work_start, work_end = working_hours

    # Step 1: normalize busy intervals
    busy = _merge_intervals(busy_intervals)
    if explain:
        explanations.append(f"Merged busy intervals: {busy}")

    # Step 2: apply buffer
    busy = _apply_buffer(busy, buffer)
    if explain:
        explanations.append(f"Applied buffer ({buffer} mins): {busy}")

    # Step 3: merge again after buffer expansion
    busy = _merge_intervals(busy)
    if explain:
        explanations.append(f"Re-merged buffered intervals: {busy}")

    # Step 4: determine search window
    search_start, search_end = work_start, work_end
    if candidate_window:
        clamped = _clamp(candidate_window, working_hours)

        if not clamped:
            if explain:
                return {
                    "slots": [],
                    "reason": "Candidate window falls outside working hours.",
                    "explanations": explanations,
                }
            return []

        search_start, search_end = clamped
        if explain:
            explanations.append(f"Using clamped candidate window: {clamped}")

    results = []
    current = search_start

    for start, end in busy:
        if end <= search_start:
            continue
        if start >= search_end:
            break

        gap_start = max(current, search_start)
        gap_end = min(start, search_end)

        if gap_end - gap_start >= duration:
            slot = (gap_start, gap_start + duration)
            results.append(slot)

            if explain:
                explanations.append(
                    f"Added slot {slot} from gap ({gap_start}, {gap_end})"
                )

            if len(results) >= max_results:
                return (
                    {"slots": results, "explanations": explanations}
                    if explain
                    else results
                )

        current = max(current, end)

    # final gap
    if current < search_end:
        if search_end - current >= duration:
            slot = (current, current + duration)
            results.append(slot)

            if explain:
                explanations.append(
                    f"Added final slot {slot} from trailing gap"
                )

    # Edge case: no results
    if not results:
        if explain:
            return {
                "slots": [],
                "reason": "No available slots meet the duration and buffer constraints.",
                "explanations": explanations,
            }
        return []

    return {"slots": results[:max_results], "explanations": explanations} if explain else results[:max_results]
